backend/src/services/orchestrator/chain.ts
import path from 'path';
import fs from 'fs';
import { toolDeps, type ToolKey } from './deps.js';
import { ensureDir } from '../../util/filesys.js';
type Missing = { missingInputs: string[]; haveAll: boolean };
export function computeMissing(targetDir:string, tool:ToolKey): Missing {
  const def = toolDeps[tool];
  const missing = def.inputs.filter(inp => !fs.existsSync(path.join(targetDir, inp)));
  return { missingInputs: missing, haveAll: missing.length === 0 };
}
// Minimal prerequisite chain for Aquatone and others
export function prereqsFor(tool:ToolKey): ToolKey[] {
  if (tool === 'aquatone' || tool === 'nuclei' || tool === 'gobuster' || tool === 'ffuf') {
    // Need live_urls.txt
    return ['httpx','dnsx','subfinder','assetfinder','findomain','chaos','amass'];
  }
  if (tool === 'httpx') return ['dnsx','subfinder','assetfinder','findomain','chaos','amass'];
  if (tool === 'dnsx') return ['subfinder','assetfinder','findomain','chaos','amass'];
  return [];
}
export function planForSingleTool(targetDir:string, tool:ToolKey): ToolKey[] {
  const out: ToolKey[] = [];
  const need = computeMissing(targetDir, tool);
  if (need.haveAll) return [tool];
  const prereqs = prereqsFor(tool);
  // Schedule minimal set in logical order
  for (const t of prereqs) {
    const d = computeMissing(targetDir, t);
    if (!d.haveAll) out.push(t);
  }
  out.push(tool);
  // De-dup while keeping order
  return out.filter((t, i, a) => a.indexOf(t) === i);
}

backend/src/services/tools/common.ts
import { spawn } from 'child_process';
import path from 'path';
import fs from 'fs';
import os from 'os';
import { APP_CONFIG } from '../../config/app.js';
import { logger } from '../../util/logger.js';

export type RunOpts = {
  cwd: string;
  stdoutPath: string;
  stderrPath: string;
  env?: Record;
  onProgress?: (p:number, txt?:string)=>void;
  docker?: {
    image: string;
    binds: { host: string; container: string }[];
    args?: string[];
    command: string[];
    name?: string;
  };
  native?: {
    cmd: string;
    args: string[];
    shell?: boolean;
  };
};

export function normalizeHostPath(p: string){
  // Docker Desktop on Windows needs absolute Windows paths like C:\... mapped
  return path.resolve(p);
}

export function runExternal(opts: RunOpts): Promise {
  return new Promise((resolve, reject) => {
    const out = fs.createWriteStream(opts.stdoutPath, { flags: 'a' });
    const err = fs.createWriteStream(opts.stderrPath, { flags: 'a' });

    let child;
    if (APP_CONFIG.dockerEnabled && opts.docker) {
      const args = ['run','--rm'];
      for (const b of opts.docker.binds) {
        args.push('-v', `${normalizeHostPath(b.host)}:${b.container}`);
      }
      if (opts.docker.args) args.push(...opts.docker.args);
      const [image, ...cmd] = [opts.docker.image, ...opts.docker.command];
      args.push(image, ...cmd);
      child = spawn('docker', args, { cwd: opts.cwd });
      logger.info({args}, 'Running docker command');
    } else if (opts.native) {
      child = spawn(opts.native.cmd, opts.native.args, { cwd: opts.cwd, shell: !!opts.native.shell });
      logger.info({cmd:opts.native.cmd, args:opts.native.args}, 'Running native command');
    } else {
      return reject(new Error('No runnable spec provided'));
    }

    child.stdout.on('data', (d) => { out.write(d); opts.onProgress?.(undefined as any, d.toString()); });
    child.stderr.on('data', (d) => { err.write(d); });
    child.on('error', (e)=> reject(e));
    child.on('close', (code)=> {
      out.end(); err.end(); resolve(code ?? -1);
    });
  });
}

export function mkStdPaths(cwd:string, tool:string){
  return {
    stdoutPath: path.join(cwd, `logs/${tool}.stdout.log`),
    stderrPath: path.join(cwd, `logs/${tool}.stderr.log`)
  };
}

      cmd: 'bash',
      args: ['-lc','amass enum -passive -df roots.txt -json amass_raw_output.json && cat amass_raw_output.json | jq -r \'.name\' > subdomains_amass.txt']

  });
}
export async function combineDedup(targetDir:string){
  const { stdoutPath, stderrPath } = mkStdPaths(targetDir, 'combine_dedup');
  const cmd = 'bash';
  const args = ['-lc', 'cat subdomains_*.txt *.com.txt 2>/dev/null | cut -d " " -f 1 | sed \'s/:\$$[0-9]\\+\$$//g\' | sed \'s#https\\?://##\' | sort -u > all_subdomains_raw.txt'];
  return runExternal({ cwd: targetDir, stdoutPath, stderrPath, native: { cmd, args } });
}
export async function resolversAndDnsx(targetDir:string){
  const { stdoutPath, stderrPath } = mkStdPaths(targetDir, 'dnsx');
  const downloadResolvers = ['-lc','wget -q -O resolvers.txt https://raw.githubusercontent.com/trickest/resolvers/main/resolvers.txt || true'].join(' ');
  await runExternal({ cwd: targetDir, ...mkStdPaths(targetDir,'resolvers'), native:{ cmd:'bash', args:['-lc', downloadResolvers]}});
  const cmd = 'bash';
  const args = ['-lc',
    "dnsx -l all_subdomains_raw.txt -r resolvers.txt -a -cname -resp -o unresolved_hosts.txt -silent && " +
    "cat unresolved_hosts.txt | grep 'CNAME' | grep 's3.amazonaws.com' | awk '{print $1}' > potential_s3_takeovers.txt && " +
    "cat unresolved_hosts.txt | awk '{print $1}' | sort -u > resolved_hosts.txt"
  ];
  return runExternal({ cwd: targetDir, stdoutPath, stderrPath, native: { cmd, args } });
}

function targetDirOf(targetId:number){
  return path.join(process.cwd(), 'data', String(targetId));
}

async function recordExecution(targetId:number, tool:string, command:string){
  const info = db.prepare('INSERT INTO executions (target_id, stage, tool, command, status, started_at) VALUES (?,?,?,?,?,datetime("now"))')
    .run(targetId, 1, tool, command, 'running');
  return Number(info.lastInsertRowid);
}
function completeExecution(id:number, status: 'completed'|'failed'){
  db.prepare('UPDATE executions SET status=?, finished_at=datetime("now") WHERE id=?').run(status, id);
}

async function runTool(targetId:number, tool:string){
  const dir = targetDirOf(targetId);
  const execId = await recordExecution(targetId, tool, tool);
  io.to(`target:${targetId}`).emit('execution:update', { executionId: execId, status: 'running', tool });
  const runner = RUNNERS[tool];
  if (!runner) { completeExecution(execId, 'failed'); return; }
  const code = await runner(dir);
  if (tool === 'httpx') await RUNNERS['derive_live_urls'](dir);
  completeExecution(execId, code === 0 ? 'completed' : 'failed');
  io.to(`target:${targetId}`).emit('execution:update', { executionId: execId, status: code === 0 ? 'completed' : 'failed', tool });
}

export async function enqueueToolsForTarget(targetId:number, tools:string[]){
  for (const t of tools) {
    await toolQueue.add('tool', { targetId, tool: t }, { removeOnComplete: true, removeOnFail: false });
  }
}

makeWorker('tools', async (job) => {
  const { targetId, tool } = job.data as { targetId:number; tool:string };
  await runTool(targetId, tool);
});

README.md
Akshay’s Framework — Windows (Docker) + BullMQ + Redis + React/Express

Prerequisites
- Windows with Docker Desktop (or Kali VM with Docker)
- Node.js 20+
- npm or pnpm
- Docker Desktop running
- Optional: aquatone.exe in PATH (for native screenshots)

Setup
1) Start Redis (Docker)
- cd akshays-framework/docker
- docker compose up -d redis

2) Backend
- cd ../backend
- npm i
- set environment (optional):
  - set DOCKER_ENABLED=1
  - set AQUATONE_CHROME_PATH=C:\Users\aksha\Documents\Docker\chrome-win\chrome.exe
  - set CHAOS_KEY=your_chaos_key
- npm run dev

3) Frontend
- cd ../frontend
- npm i
- npm run dev
- Open http://localhost:5173

Basic flow
- Create a target (single domain or paste contents for roots.txt).
- Full Recon: runs Stage1→Stage2→Stage3 subset (subfinder/assetfinder/findomain/chaos/amass → dnsx → httpx → live_urls.txt → aquatone → katana).
- Single Tool: open target → Execute on Aquatone (auto-runs prerequisites if live_urls.txt missing).

Smart dependencies (single-tool example)
- Aquatone will compute:
